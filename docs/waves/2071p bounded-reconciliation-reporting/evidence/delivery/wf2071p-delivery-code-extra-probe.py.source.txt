def report_tree_termination():

  body='import subprocess,sys,time\nfrom pathlib import Path\np=subprocess.Popen([sys.executable,"-c","import time;time.sleep(20)"])\nPath('+repr(str(pidfile))+').write_text(str(p.pid))\ntime.sleep(20)\n'
  worker=write(root,'worker.py',body)
  with mock.patch.object(upgrade,'__file__',str(worker)),mock.patch.object(upgrade,'_preferred_python',return_value=sys.executable):
   result=upgrade._collect_reconciliation_report(root,budget_s=.35)
  assert result['reconciliation_state']=='incomplete' and result['reconciliation_reason']=='reporting_timeout'
  assert pidfile.exists(),'worker startup not reached'
  pid=int(pidfile.read_text());alive=True
  for attempt in range(20):
   try: os.kill(pid,0)
   except ProcessLookupError: alive=False;break
   # A terminated orphan may remain visible as a zombie briefly; ps state
   # distinguishes it from an owned descendant still doing work.
   state=subprocess.run(['ps','-o','stat=','-p',str(pid)],capture_output=True,text=True).stdout.strip()
   if not state or state.startswith('Z'):alive=False;break
   time.sleep(.025)
  if alive:
   os.kill(pid,9)
  assert not alive,'owned grandchild remained alive after parent reporting timeout'
  return {'worker_pid_persisted':True,'grandchild_pid':pid,'grandchild_running_after_timeout':alive,'state':result['reconciliation_state'],'elapsed':result['reconciliation_elapsed']}
record('owned-report-worker-grandchild-timeout',report_tree_termination)
